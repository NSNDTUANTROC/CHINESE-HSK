function checkAnswer(index) { const inputEl = document.getElementById(`input_${index}`); if (!inputEl) return; const inputVal = inputEl.value.trim().toLowerCase(); const mode = document.getElementById('modeSelect').value; const item = currentQuizData[index]; const statusTd = document.getElementById(`status_${index}`); const rowTr = document.getElementById(`row_${index}`); if (inputVal === '') { statusTd.innerText = '-'; statusTd.className = 'col-status'; rowTr.className = ''; updateStats(); return; } let isCorrect = false; // Hàm hỗ trợ tách đáp án theo các dấu phân cách / , ; hoặc - const splitOptions = (text) => { if (!text) return []; return text.split(/[\/,;]/) .flatMap(part => part.split(/\s+hoặc\s+/i)) .map(s => s.trim().toLowerCase()) .filter(s => s.length > 0); }; if (mode === 'pinyin_to_chinese') { const pinyinOptions = splitOptions(item.pinyin); const chineseOptions = splitOptions(item.chinese); // Kiểm tra xem người dùng gõ trùng với BẤM KỲ đáp án Pinyin hoặc Chữ Hán nào if (pinyinOptions.includes(inputVal) || chineseOptions.includes(inputVal)) { isCorrect = true; } } else { const meaningOptions = splitOptions(item.meaning); // Chế độ gõ tiếng Việt: đúng nếu khớp với 1 đáp án hoặc nằm trong cụm nghĩa isCorrect = meaningOptions.some(option => option === inputVal || (inputVal.length >= 2 && option.includes(inputVal)) ); } if (isCorrect) { statusTd.innerText = '✓ Đúng'; statusTd.style.color = 'var(--success)'; rowTr.className = 'row-correct'; } else { statusTd.innerText = '✗ Sai'; statusTd.style.color = 'var(--danger)'; rowTr.className = 'row-wrong'; } // Hiển thị đầy đủ thông tin sau khi trả lời document.getElementById(`chinese_${index}`).innerText = item.chinese; document.getElementById(`pinyin_${index}`).innerText = item.pinyin; document.getElementById(`meaning_${index}`).innerText = item.meaning; updateStats(); }
