// MÃ KHÓA BẢO MẬT CHÍNH XÁC TỪ Ô Z1 CỦA BẠN const SECRET_KEY = "CAMTU2019S2THANHAN2025"; // ------------------------------------------------------------------------- // ĐỌC FILE EXCEL & KIỂM TRA MÃ KHÓA Ở CỘT Z (Ô Z1) // ------------------------------------------------------------------------- document.getElementById('excelFile').addEventListener('change', function(e) { const file = e.target.files[0]; if (!file) return; const statusText = document.getElementById('loadingStatus'); if (statusText) { statusText.innerText = "⏳ Đang kiểm tra mã khóa..."; statusText.style.color = "var(--primary)"; } const reader = new FileReader(); reader.onload = function(e) { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: 'array' }); const worksheet = workbook.Sheets[workbook.SheetNames[0]]; // 1. Lấy giá trị trực tiếp từ ô Z1 trong Excel const cellZ1 = worksheet['Z1'] ? worksheet['Z1'].v.toString().trim() : ''; // 2. Chuyển sheet thành mảng rows để phòng trường hợp đọc theo hàng const rows = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: "" }); if (!rows || rows.length < 2) { alert("File Excel không có dữ liệu!"); return; } // Kiểm tra xem ô Z1 hoặc dòng 1 có chứa đúng mã khóa hay không let hasValidKey = false; if (cellZ1 === SECRET_KEY) { hasValidKey = true; } else if (rows[0]) { // Kiểm tra quét trên hàng 1 (bao gồm cột Z - chỉ số 25) for (let col = 0; col < rows[0].length; col++) { if (rows[0][col] && rows[0][col].toString().trim() === SECRET_KEY) { hasValidKey = true; break; } } } if (!hasValidKey) { alert("❌ TRUY CẬP BỊ TỪ CHỐI!\n\nFile Excel này KHÔNG HỢP LỆ hoặc đã bị chỉnh sửa làm mất/sai mã xác thực ở ô Z1."); document.getElementById('excelFile').value = ''; if (statusText) statusText.innerText = ""; return; } // 3. Tìm dòng tiêu đề let headerRowIndex = 0; for (let i = 0; i < Math.min(rows.length, 5); i++) { const rowStr = rows[i].join(" ").toUpperCase(); if (rowStr.includes("CHENESE") || rowStr.includes("CHINESE") || rowStr.includes("PINYIN") || rowStr.includes("MEANING")) { headerRowIndex = i; break; } } const headers = rows[headerRowIndex].map(h => h.toString().trim().toUpperCase()); let colChinese = -1, colType = -1, colPinyin = -1, colMeaning = -1; headers.forEach((h, idx) => { if (h.includes('INPUT')) return; if ((h.includes('CHENESE') || h.includes('CHINESE') || h.includes('HÁN')) && colChinese === -1) colChinese = idx; else if (h.includes('TYPE') || h.includes('LOẠI TỪ')) colType = idx; else if (h.includes('PINYIN') || h.includes('PHIÊN ÂM')) colPinyin = idx; else if (h.includes('MEANING') || h.includes('TIẾNG VIỆT') || h.includes('NGHĨA')) colMeaning = idx; }); if (colChinese === -1) colChinese = 1; if (colType === -1) colType = 2; if (colPinyin === -1) colPinyin = 3; if (colMeaning === -1) colMeaning = 4; const parsedData = []; for (let i = headerRowIndex + 1; i < rows.length; i++) { const row = rows[i]; if (!row || row.length === 0) continue; const chinese = row[colChinese] ? row[colChinese].toString().trim() : ''; const type = row[colType] ? row[colType].toString().trim() : ''; const pinyin = row[colPinyin] ? row[colPinyin].toString().trim() : ''; const meaning = row[colMeaning] ? row[colMeaning].toString().trim() : ''; if (chinese || meaning) { parsedData.push({ chinese, type, pinyin, meaning }); } } if (parsedData.length === 0) { alert("Không tìm thấy dữ liệu từ vựng hợp lệ!"); return; } fullVocabData = parsedData; localStorage.setItem('hsk_vocab_data', JSON.stringify(fullVocabData)); if (statusText) { statusText.innerText = `✅ Đã nạp thành công ${fullVocabData.length} từ!`; statusText.style.color = "var(--success)"; } document.getElementById('rangeFrom').value = 1; document.getElementById('rangeTo').value = fullVocabData.length; sequentialOffset = 0; startNewQuiz(); } catch (err) { alert("Lỗi khi đọc file Excel: " + err.message); } }; reader.readAsArrayBuffer(file); });
